in web console of browser:
ws.send('command in single quotes')

create user:
{"commandType":"CLIENT_REGISTER_USER","command":{"userName":"user1"}}

fetch all users:
{"commandType":"CLIENT_FETCH_ALL_USERS","command":{"userId":1}}

self identify:
{"commandType":"CLIENT_SELF_IDENTIFY","command":{"userId":1}}

create group chat
{"commandType":"CLIENT_CREATE_CHAT","command":{"chatName":"chat group 1","userIds":[1,2,3]}}

send message to chat
{"commandType":"CLIENT_SEND_MESSAGE_TO_CHAT","command":{"chatId":1,"content":"user 1 welcomes you to the group"}}

acknowledge message receive
{"commandType":"CLIENT_ACKNOWLEDGE_MESSAGE_RECEIVED","command":{"messageId":1,"chatId":1}}

fetch all pending messages
{"commandType":"CLIENT_FETCH_ALL_PENDING_MESSAGES","command":{"userId":3}}
